// Quering data in database using Django shell (we can also use python files):
// run : python manage.py shell

// example:
from notes.models import Notes
mynote = Notes.objects.get(pk='1') # pk stands for primary key (id's are auto incremented starting from 1 and created by default), if we delete an object they'll continue adding up
print(mynote.title)

// we can query all of the notes by using .all() and then convert the QuerySet into a list and get their data
all_notes = list(Notes.objects.all())
all_notes[0].title

// we can also use filters:
Notes.objects.filter(title__startswith="second") # this will also return a Queryset
Notes.objects.filter(title__icontains="note")
// the opposite of filter is .exclude, which will return the objects, that doesn't have the property we give it
Notes.objects.exclude(title__icontains="second")

// we can also chain to make complex queries
Notes.objects.filter(text__icontains="note").exclude(title__icontains="second")
